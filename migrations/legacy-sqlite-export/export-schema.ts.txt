/** Export the current schema without opening or copying the user's database. */
import fs from "node:fs";
import path from "node:path";

async function main() {
  process.env.NEXT_PHASE = "phase-production-build";
  const { db } = await import("../src/lib/db");
  const objects = db.prepare("SELECT type, name, sql FROM sqlite_master WHERE sql IS NOT NULL AND name NOT LIKE 'sqlite_%' ORDER BY CASE type WHEN 'table' THEN 0 ELSE 1 END, rowid").all() as { type: string; name: string; sql: string }[];
  const schema = objects.map(object => ({ ...object, columns: object.type === "table" ? db.prepare(`PRAGMA table_info("${object.name}")`).all() : undefined }));
  fs.mkdirSync(path.resolve("migrations"), { recursive: true });
  fs.writeFileSync(path.resolve("migrations/sqlite-schema.json"), JSON.stringify(schema, null, 2) + "\n");
  console.log(`Exported ${objects.filter(o => o.type === "table").length} tables and ${objects.filter(o => o.type === "index").length} indexes; no user data exported.`);
  db.close();
}
main().catch(error => { console.error(error); process.exitCode = 1; });
